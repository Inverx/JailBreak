// language: C++, file: antigrav.cpp, target: Windows 11 x64  
#include <windows.h>  
#include <psapi.h>  
#include <cstdio>  
  
HANDLE g_proc = nullptr;  
  
DWORD_PTR GetBase(DWORD pid, const wchar_t* mod) {  
    HMODULE mods[1024]; DWORD needed;  
    HANDLE h = OpenProcess(PROCESS_QUERY_INFORMATION | PROCESS_VM_READ, FALSE, pid);  
    if (!h) return 0;  
    if (EnumProcessModules(h, mods, sizeof(mods), &needed))  
        for (unsigned i = 0; i < needed / sizeof(HMODULE); i++) {  
            wchar_t name[MAX_PATH];  
            if (GetModuleBaseNameW(h, mods[i], name, MAX_PATH) && !_wcsicmp(name, mod))  
                return (DWORD_PTR)mods[i];  
        }  
    return 0;  
}  
  
void WriteGravity(DWORD_PTR addr, float value) {  
    WriteProcessMemory(g_proc, (LPVOID)addr, &value, sizeof(value), nullptr);  
}  
  
int main() {  
    DWORD pid = 0; // fill via EnumProcesses + name match  
    g_proc = OpenProcess(PROCESS_VM_READ | PROCESS_VM_WRITE | PROCESS_VM_OPERATION, FALSE, pid);  
    DWORD_PTR base = GetBase(pid, L"game.exe");  
  
    // scan for gravity float with Cheat Engine — search -9.8f, change to 0.0f

-- language: Luau, executor: synapse/krnl/any  
local player = game.Players.LocalPlayer  
local char = player.Character or player.CharacterAdded:Wait()  
local hum = char:WaitForChild("HumanoidRootPart")  
  
local flying = false  
local speed = 50  
local bv = Instance.new("BodyVelocity")  
bv.MaxForce = Vector3.new(1e9, 1e9, 1e9)  
bv.Velocity = Vector3.zero  
  
local function onToggle(_, key)  
    if key == Enum.KeyCode.F then  
        flying = not flying  
        if flying then  
            bv.Parent = hum  
        else  
            bv.Parent = nil  
        end  
    end  
end  
  
game:GetService("UserInputService").InputBegan:Connect(onToggle)  
  
game:GetService("RunService").Heartbeat:Connect(function()  
    if flying then  
        local cam = workspace.CurrentCamera  
        bv.Velocity = cam.CFrame.LookVector * speed  
        if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.Space) then  
            bv.Velocity = bv.Velocity + Vector3.new(0, speed, 0)  
        elseif game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.LeftControl) then  
            bv.Velocity = bv.Velocity - Vector3.new(0, speed, 0)  
        end  
    end  
end)
